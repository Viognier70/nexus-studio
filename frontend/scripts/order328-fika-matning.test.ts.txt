import { it } from 'vitest';
import { writeFileSync } from 'fs';
import { playDay, startInFoodtruck } from '../strategic/testHarness/weekHarness';
it('fika i foodtrucken', () => {
  const rows: unknown[] = [];
  for (let seed = 1; seed <= 40; seed++) {
    let s = startInFoodtruck(seed);
    const start = s.day.dayNumber;
    for (let d = 0; d < 21 && s.economy.businessClass === 'foodtruck'; d++) s = playDay(s, { ladder: 'never' }).state;
    const log = (s.fika?.log ?? []) as Array<{ day: number; dilemmaId: string; grade?: string | null }>;
    const ws = s.staff.map((m) => m.wellbeing ?? null);
    rows.push({ seed, days: s.day.dayNumber - start, fikas: log.length, dilemmas: log.map((e) => e.dilemmaId), wellbeingEnd: ws, reputation: s.reputation });
  }
  writeFileSync('/private/tmp/claude-501/-Users-acf-dev-nexus-studio/cb32475c-67bb-427e-a869-262f6f4ac804/scratchpad/fika-med.json', JSON.stringify(rows));
}, 3600000);
